\section{Requisitos del sistema}
    \subsection{¿Qué debe hacer el sistema?}
        El sistema debe permitir al usuario seleccionar un número del 0 al 9 desde una página web y mostrarlo en un display de 7 segmentos conectado a un Arduino.

    \subsection{¿Qué entradas tendrá?}
        \begin{itemize}
            \item Selección de número (0-9) desde la página web
            \item Selección de apagado desde la página web
            \item Entrada de datos por medio de comunicación serial hacia el Arduino
        \end{itemize}
        
    \subsection{¿Qué salidas generá?}
        \begin{itemize}
            \item Visualización del número seleccionado en la página web
            \item Visualización del número en el display de 7 segmentos
        \end{itemize}


\section{Léxico del sistema}
    \subsection{Arduino}
    Microcontrolador que recibe datos y controla el display.

    \subsection{Display de 7 segmentos}
    Dispositivo electrónico que muestra números mediante LEDs.

    \subsection{Comunicación Serial}
    Método de transmisión de datos entre el servidor y Arduino (9600 baudios).

    \subsection{Servidor}
    Programa que recibe datos de la web y los envía al Arduino.

    \subsection{Baudios}
    Velocidad de transmición de datos (bits por segundo).


\section{Modelo Entrada-Proceso-Salida}
    \subsection{Entrada}
    Usuario selecciona un número en la página web

    \subsection{Proceso}
        \begin{enumerate}
            \item La web envía el número al servidor
            \item El servidor lo procesa
            \item El servidor lo envía por serial al Arduino
            \item Arduino interpreta el número
            \item Arduino activa los segmentos correspondientes
        \end{enumerate}
    
    \subsection{Salida}
        \begin{itemize}
            \item Número mostrado en la interfaz web
            \item Número mostrado en el display físico
        \end{itemize}

\section{Conexión del display al Arduino}
    \begin{itemize}
        \item Cada segmento (a-g) se conecta a un pin dígital (ej: a=2, b=3..., g=8)
        \item Se usan resistencias (470 $\Omega$)
        \item El cátodo común va a GND
    \end{itemize}

\section{Verificación}
